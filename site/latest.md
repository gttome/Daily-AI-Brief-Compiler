<a href="{{ '/stories/2026-11-09/nov09-t1-command-permissions/' | relative_url }}" data-item-id="dab-story-2026-11-09-m01" data-edition-date="2026-11-09" data-action="permanent_page_clicks">Open the permanent story page</a>
